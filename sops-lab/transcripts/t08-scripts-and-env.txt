# exec-env decrypts in memory and runs a command with the values

# in its environment. Nothing decrypted touches the disk.

$ sops exec-env secrets.enc.env 'printenv DB_PASSWORD'
env-fixture-password-3141

$ sops exec-env secrets.enc.env 'printenv PLAIN_NOTE'
this value was never secret

# The surrounding shell never sees them:

$ printenv DB_PASSWORD; echo exit=$?
exit=1

# exec-file is the file-shaped version: the decrypted document

# appears as a temp file whose path replaces {} in the command,

# and is shredded when the command ends.

$ sops exec-file app.enc.yaml 'grep host {}'
    host: db.example.com

# And the plain pipe for everything else: decrypt to stdout,

# handed to whatever reads the format.

$ sops decrypt app.enc.yaml | grep -E 'host|port'
    host: db.example.com
    port: 5432
