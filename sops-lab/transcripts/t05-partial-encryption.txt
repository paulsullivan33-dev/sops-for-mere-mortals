# Sometimes only some values are secret. --encrypted-regex seals

# only the values whose KEY matches the pattern.

$ cat partial.yaml
service_name: fixture-web
public_url: https://www.example.com
db_password: partial-fixture-secret-777
api_token: partial-fixture-token-888
retry_count: 3

$ sops encrypt --age age1svvqr3y4rhe0dsypnrm5qa8fwlauav7f9ze5hlgm08kysczr450qmg4zht --encrypted-regex ^(db_password|api_token)$ --output partial.enc.yaml partial.yaml
